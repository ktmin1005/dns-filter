cmake_minimum_required(VERSION 3.16)
project(dnsfilter CXX)
set(CMAKE_CXX_STANDARD 20)
add_compile_options(-Wall -Wextra -g -fsanitize=undefined)
add_link_options(-fsanitize=undefined)
find_package(Threads REQUIRED)
add_executable(dnsfilter src/main.cpp src/blocklist.cpp src/dns_parser.cpp src/access_control.cpp)
target_link_libraries(dnsfilter PRIVATE Threads::Threads)
